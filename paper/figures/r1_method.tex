   \textcolor{rOneShift!75!black}{\textbf{SHIFT}}: matched, centre ${>}0.3$\,m or heading ${>}10^\circ$\\[3pt]
   effect $\to$ 6.4\,m tile $u$ of its box centre (either version)\\[2pt] $\Rightarrow$ unit $(i,u)$};

%---------------------------------------------------------------- step 4
\node[step, text width=3.8cm, minimum height=5.0cm] (ray) at (10.6,0)
  {\rOneStep{4} \textbf{Physical verification} (receiver's rays)\\[1pt]

   \textcolor{rOneSup!80!black}{\textbf{SUP}} ends inside, \textcolor{rOneMal}{\textbf{TRAV}} passes through
   (ends before: ignored)\\[3pt]
   $N=\mathrm{SUP}{+}\mathrm{TRAV}<3$ or own-vehicle box\\ $\Rightarrow$ \textbf{UNKNOWN} $=$ keep\\[3pt]
   else Wilson-95 $[\mathrm{lo},\mathrm{hi}]$ on $s=\mathrm{SUP}/N$:\\[1pt]
   \setlength{\tabcolsep}{2.5pt}%
   \begin{tabular}{@{}lll@{}}
     & \textit{gate} & \textit{useful}\\
     ADD    & $\mathrm{hi}(s^{+}){<}0.5$ & $\mathrm{lo}(s^{+}){>}0.5$\\
     REMOVE & $\mathrm{lo}(s^{-}){>}0.5$ & $\mathrm{hi}(s^{-}){<}0.5$
   \end{tabular}\\[1pt]
   SHIFT: $I^{+}\cap I^{-}=\emptyset$ and $s^{+}{<}s^{-}$ $\Rightarrow$ gate,\\ $s^{+}{>}s^{-}$ $\Rightarrow$ useful; otherwise \emph{none}};

\caption{R1 per receiver frame. The frozen AttFuse detector decodes once with all agents
($Y^{+}$, step~1) and once without each sender ($Y^{-}(i)$, step~2). Their differences are ADD, REMOVE and
SHIFT effects, each attributed to a sender $i$ and a 6.4\,m tile $u$ (step~3), and every effect box is checked
against the receiver's own LiDAR rays with a Wilson-95 resolvability margin; too few rays or the receiver's own
box give UNKNOWN, which keeps the sender (step~4, inset). Sender $i$ is switched off in tile $u$ only if an
effect resolvably contradicts the rays and none is resolvably confirmed (step~5), and the output is composed
tile by tile from the decodes already computed, before NMS, together with a per-(sender, tile) reliability
record (step~6).}
\label{fig:r1method}
